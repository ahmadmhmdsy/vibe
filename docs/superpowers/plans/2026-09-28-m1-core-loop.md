# M1 Core Loop Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Prove the vibe-coding core loop end-to-end — a single chat session where the agent writes a Vite + React + TS app, installs deps, builds, runs, previews, and auto-fixes build/runtime errors.

**Architecture:** A FastAPI backend (services/vibe) exposes capability seams — `AgentRuntime`, `LlmProvider`, `Executor`, `ToolRegistry`, `SessionStore`, `EventBus` — composed over an append-only per-session EventLog. A React + TypeScript frontend (apps/vibe) renders chat/preview/code/sessions/settings from the log over HTTP + SSE. Auto-run is a policy that composes runtime calls, never a branch inside the loop.

**Tech Stack:** Python 3.13 · FastAPI · LangGraph · Pydantic v2 · SQLite (stdlib) · asyncio subprocesses · Playwright (headless Chromium) · pytest. Frontend: Vite · React 19 · TypeScript · zustand · Monaco · vitest.

**Spec:** `docs/superpowers/specs/2026-09-28-vibe-coding-app-design.md`

## Global Constraints

- **Python 3.13** (uv-managed venv at `services/vibe/.venv`); **Node 24 / pnpm** for the frontend. Never mix package managers.
- **Secrets server-side only.** LLM API keys live in `services/vibe/.env` (gitignored); the frontend sees only provider names + masked key status.
- **Workspace confinement.** Agent file tools are restricted to `~/vibe-workspaces/<session-id>/`; path allowlists reject anything outside.
- **Command allowlist.** Executor runs only `pnpm install`, `vite build`, `vite dev` with fixed flags — no arbitrary shell.
- **Timeouts + truncation** on every executor method; dev server has an idle timeout.
- **Model-visible means logged.** Anything the agent sees is reconstructable from the EventLog.
- **Validation labels:** PASS / FAIL / SKIPPED / BLOCKED / NEEDS_USER_DECISION.
- **Task statuses:** pending / in_progress / done / blocked.
- **App name:** working name "vibe" (`apps/vibe` + `services/vibe`).

# services/vibe/src/vibe/events.py
from __future__ import annotations
import asyncio
from dataclasses import dataclass, field
from typing import Any, AsyncIterator

@dataclass
class Event:
    session_id: str
    seq: int
    type: str
    payload: dict[str, Any]

class EventLog:
    """In-memory append-only log. SQLite persistence lands in Task 4."""
    def __init__(self):
        self._events: dict[str, list[Event]] = {}
        self._seq: dict[str, int] = {}

    async def append(self, session_id: str, type: str, payload: dict[str, Any]) -> int:
        seq = self._seq.get(session_id, 0) + 1
        self._seq[session_id] = seq
        event = Event(session_id=session_id, seq=seq, type=type, payload=payload)
        self._events.setdefault(session_id, []).append(event)
        return seq

    async def get(self, session_id: str, after_seq: int = 0) -> list[Event]:
        return [e for e in self._events.get(session_id, []) if e.seq > after_seq]

class EventBus:
    def __init__(self):
        self._queues: dict[str, asyncio.Queue[Event]] = {}

    def _queue(self, session_id: str) -> asyncio.Queue[Event]:
        return self._queues.setdefault(session_id, asyncio.Queue())

    async def publish(self, session_id: str, event: Event) -> None:
        await self._queue(session_id).put(event)

    async def subscribe(self, session_id: str) -> AsyncIterator[Event]:
        q = self._queue(session_id)
        try:
            while True:
                event = await q.get()
                yield event
        except asyncio.CancelledError:
            raise
```

# services/vibe/src/vibe/store.py
from __future__ import annotations
import sqlite3, uuid, json
from dataclasses import dataclass
from datetime import datetime, timezone
from pathlib import Path
from vibe.models import ModelRef

@dataclass
class Session:
    id: str
    title: str
    model_ref: ModelRef
    created_at: str
    updated_at: str

class SessionStore:
    def __init__(self, db_path: Path):
        self._db = sqlite3.connect(db_path, check_same_thread=False)
        self._db.row_factory = sqlite3.Row
        self._migrate()

    def _migrate(self):
        self._db.executescript("""
            CREATE TABLE IF NOT EXISTS sessions (
                id TEXT PRIMARY KEY,
                title TEXT NOT NULL,
                model_ref TEXT NOT NULL,
                created_at TEXT NOT NULL,
                updated_at TEXT NOT NULL
            );
        """)
        self._db.commit()

    async def create_session(self, model_ref: ModelRef) -> str:
        sid = uuid.uuid4().hex[:12]
        now = datetime.now(timezone.utc).isoformat()
        title = f"Untitled {sid[:4]}"
        self._db.execute(
            "INSERT INTO sessions (id, title, model_ref, created_at, updated_at) VALUES (?, ?, ?, ?, ?)",
            (sid, title, json.dumps({"provider_id": model_ref.provider_id, "model_id": model_ref.model_id}), now, now),
        )
        self._db.commit()
        return sid

    async def list_sessions(self) -> list[Session]:
        rows = self._db.execute("SELECT * FROM sessions ORDER BY updated_at DESC").fetchall()
        return [self._to_session(r) for r in rows]

    async def get_session(self, session_id: str) -> Session | None:
        r = self._db.execute("SELECT * FROM sessions WHERE id = ?", (session_id,)).fetchone()
        return self._to_session(r) if r else None

    async def rename_session(self, session_id: str, title: str) -> None:
        now = datetime.now(timezone.utc).isoformat()
        self._db.execute("UPDATE sessions SET title = ?, updated_at = ? WHERE id = ?", (title, now, session_id))
        self._db.commit()

    async def delete_session(self, session_id: str) -> None:
        self._db.execute("DELETE FROM sessions WHERE id = ?", (session_id,))
        self._db.commit()

    def _to_session(self, r: sqlite3.Row) -> Session:
        mr = json.loads(r["model_ref"])
        return Session(
            id=r["id"], title=r["title"],
            model_ref=ModelRef(provider_id=mr["provider_id"], model_id=mr["model_id"]),
            created_at=r["created_at"], updated_at=r["updated_at"],
        )
```

# services/vibe/src/vibe/runtime/langgraph.py
from __future__ import annotations
from typing import AsyncIterator, Callable
from langgraph.graph import StateGraph, MessagesState, START
from langgraph.prebuilt import ToolNode, tools_condition
from langchain_core.messages import AIMessage
from vibe.models import AgentInput, AgentEvent, Message
from vibe.runtime.base import AgentContext

def _to_lc_message(m: Message):
    from langchain_core.messages import HumanMessage, AIMessage, SystemMessage, ToolMessage
    if m.role == "user":
        return HumanMessage(content=m.content)
    if m.role == "assistant":
        return AIMessage(content=m.content, tool_calls=m.tool_calls)
    if m.role == "system":
        return SystemMessage(content=m.content)
    return ToolMessage(content=m.content, tool_call_id=m.tool_call_id, name=m.name)

class LangGraphRuntime:
    def __init__(self, llm_factory: Callable, tool_registry):
        self._llm_factory = llm_factory
        self._tool_registry = tool_registry

    def _build_graph(self, model_ref, ctx: AgentContext):
        llm = self._llm_factory(model_ref)
        lc_tools = self._tool_registry.to_langchain_tools(ctx)
        llm_with_tools = llm.bind_tools(lc_tools)

        def call_model(state: MessagesState):
            response = llm_with_tools.invoke(state["messages"])
            return {"messages": [response]}

        tool_node = ToolNode(lc_tools)
        graph = StateGraph(MessagesState)
        graph.add_node("call_model", call_model)
        graph.add_node("execute_tools", tool_node)
        graph.add_edge(START, "call_model")
        graph.add_conditional_edges("call_model", tools_condition)
        graph.add_edge("execute_tools", "call_model")
        return graph.compile()

    async def run(self, input: AgentInput, ctx: AgentContext) -> AsyncIterator[AgentEvent]:
        app = self._build_graph(input.model, ctx)
        messages = [_to_lc_message(m) for m in input.messages]
        async for chunk in app.astream({"messages": messages}, stream_mode="updates"):
            for node_name, node_output in chunk.items():
                if node_name == "call_model":
                    msg = node_output["messages"][-1]
                    if msg.content:
                        yield AgentEvent(kind="text_chunk", data={"text": msg.content})
                    for tc in getattr(msg, "tool_calls", []):
                        yield AgentEvent(kind="tool_call", data={"tool_call": tc})
                elif node_name == "execute_tools":
                    for tm in node_output["messages"]:
                        yield AgentEvent(kind="tool_result", data={
                            "tool_call_id": tm.tool_call_id, "name": tm.name,
                            "content": tm.content, "is_error": False})
        yield AgentEvent(kind="turn_complete", data={})

# services/vibe/src/vibe/tools/registry.py
from __future__ import annotations
from typing import Any, Callable, Awaitable
from vibe.models import ToolSchema, ToolResult
from vibe.runtime.base import AgentContext

Handler = Callable[[dict[str, Any], AgentContext], Awaitable[ToolResult]]

class ToolRegistry:
    def __init__(self):
        self._tools: dict[str, tuple[ToolSchema, Handler]] = {}

    def register(self, name: str, schema: ToolSchema, handler: Handler) -> None:
        self._tools[name] = (schema, handler)

    def schemas(self) -> list[ToolSchema]:
        return [s for s, _ in self._tools.values()]

    async def execute(self, name: str, arguments: dict[str, Any], ctx: AgentContext) -> ToolResult:
        if name not in self._tools:
            return ToolResult(tool_call_id="", name=name, content=f"unknown tool: {name}", is_error=True)
        _, handler = self._tools[name]
        return await handler(arguments, ctx)

    def to_langchain_tools(self, ctx: AgentContext) -> list:
        from langchain_core.tools import Tool
        tools = []
        for name, (schema, handler) in self._tools.items():
            tools.append(Tool.from_function(
                name=name, description=schema.description,
                coroutine=lambda args, h=handler, c=ctx: h(args, c),
                args_schema=schema.parameters,
            ))
        return tools

from __future__ import annotations
import asyncio, socket
from pathlib import Path
from vibe.models import ToolResult
from vibe.executor.base import DevServerHandle

def _free_port() -> int:
    s = socket.socket()
    s.bind(("127.0.0.1", 0))
    port = s.getsockname()[1]
    s.close()
    return port

class LocalSubprocessExecutor:
    def __init__(self, workspace: Path, timeout: int = 120):
        self._workspace = workspace
        self._timeout = timeout
        self._dev_server: DevServerHandle | None = None

    async def _run(self, cmd: list[str]) -> ToolResult:
        try:
            proc = await asyncio.create_subprocess_exec(
                *cmd, cwd=self._workspace,
                stdout=asyncio.subprocess.PIPE, stderr=asyncio.subprocess.STDOUT)
            out, _ = await asyncio.wait_for(proc.communicate(), timeout=self._timeout)
            text = out.decode(errors="replace")[:20000]
            return ToolResult("", cmd[0], text, proc.returncode != 0)
        except asyncio.TimeoutError:
            return ToolResult("", cmd[0], "timeout", True)
        except Exception as e:
            return ToolResult("", cmd[0], str(e), True)

    async def install(self, packages: list[str]) -> ToolResult:
        return await self._run(["pnpm", "install", *packages])

    async def build(self) -> ToolResult:
        return await self._run(["pnpm", "run", "build"])

    async def start_dev_server(self) -> DevServerHandle:
        port = _free_port()
        proc = await asyncio.create_subprocess_exec(
            "pnpm", "run", "dev", "--port", str(port), "--host", "127.0.0.1",
            cwd=self._workspace, stdout=asyncio.subprocess.PIPE, stderr=asyncio.subprocess.STDOUT)
        self._dev_server = DevServerHandle(port=port, url=f"http://127.0.0.1:{port}/", process=proc)
        return self._dev_server

    async def stop_dev_server(self) -> None:

    async def gen():
        import asyncio
        # replay missed events

        # live stream

            async for e in bus.subscribe(sid):

async def send_message(sid: str, body: dict, request: Request):
    store = request.app.state.store
    s = await store.get_session(sid)
    if not s: raise HTTPException(404, "not found")
    # append user message to log
    await request.app.state.event_log.append(sid, "user/message", {"content": body["content"]})
    # build context
    workspace = Path(request.app.state.settings.workspace_dir) / sid
    workspace.mkdir(parents=True, exist_ok=True)
    executor = LocalSubprocessExecutor(workspace=workspace)
    ctx = AgentContext(session_id=sid, workspace_path=workspace, executor=executor,
                       event_bus=request.app.state.event_bus, store=store)
    inp = AgentInput(session_id=sid, messages=[Message(role="user", content=body["content"])], model=s.model_ref)
    # run agent, collecting events
    text_parts = []
    async for event in request.app.state.runtime.run(inp, ctx):
        if event.kind == "text_chunk":
            text_parts.append(event.data.get("text", ""))
        # broadcast durable events
        if event.kind in ("text_chunk", "tool_call", "tool_result", "turn_complete"):
            await request.app.state.event_bus.publish(sid, __import__("vibe.events", fromlist=["Event"]).Event(
                session_id=sid, seq=0, type=event.kind, payload=event.data))
    # append assistant message
    content = "".join(text_parts)
    await request.app.state.event_log.append(sid, "assistant/message", {"content": content})
    return {"content": content}

# services/vibe/src/vibe/autorun/policy.py
from __future__ import annotations
from vibe.models import AgentInput, Message
from vibe.runtime.base import AgentContext

class AutoRunPolicy:
    def __init__(self, executor, runtime, max_fix_iterations: int = 3):
        self._executor = executor
        self._runtime = runtime
        self._max_fix = max_fix_iterations

    async def maybe_run(self, ctx: AgentContext, changed_files: list[str]) -> None:
        if not changed_files:
            return
        for iteration in range(self._max_fix + 1):
            build = await self._executor.build()
            if build.is_error:
                if iteration < self._max_fix:
                    await self._fix_turn(ctx, f"Build failed: {build.content}")
                continue
            # build ok — start dev server and check runtime errors
            handle = await self._executor.start_dev_server()
            logs = await self._executor.read_console_logs(tail=50)
            if logs.strip():
                if iteration < self._max_fix:
                    await self._fix_turn(ctx, f"Runtime errors: {logs}")
                continue
            return  # clean

    async def _fix_turn(self, ctx: AgentContext, error: str) -> None:
        inp = AgentInput(session_id=ctx.session_id,
                         messages=[Message(role="user", content=f"The app failed: {error}. Fix it.")],
                         model=ctx.store.get_session(ctx.session_id).model_ref if False else None)
        # model_ref resolved by caller; for M1 pass via ctx extension
        async for _ in self._runtime.run(inp, ctx):
            pass

# services/vibe/src/vibe/usage/tracker.py
from __future__ import annotations
from dataclasses import dataclass, field
from vibe.models import UsageRecord
from vibe.events import EventLog, EventBus

@dataclass
class UsageAggregate:
    input_tokens: int = 0
    output_tokens: int = 0
    cached_tokens: int = 0
    total_tokens: int = 0
    cost_usd: float = 0.0
    steps: int = 0

class UsageTracker:
    def __init__(self, log: EventLog, bus: EventBus):
        self._log = log
        self._bus = bus

    async def record(self, session_id: str, usage: UsageRecord) -> None:
        await self._log.append(session_id, "usage", usage.__dict__)
        from vibe.events import Event
        events = await self._log.get(session_id)
        seq = events[-1].seq if events else 0
        await self._bus.publish(session_id, Event(session_id=session_id, seq=seq, type="usage", payload=usage.__dict__))

    async def session_total(self, session_id: str) -> UsageAggregate:
        total = UsageAggregate()
        for e in await self._log.get(session_id):
            if e.type == "usage":
                u = UsageRecord(**e.payload)
                total.input_tokens += u.input_tokens
                total.output_tokens += u.output_tokens
                total.cached_tokens += u.cached_tokens
                total.total_tokens += u.total_tokens
                total.cost_usd += u.cost_usd
                total.steps += 1
        return total
```

export interface StreamEvent {
  seq: number;
  type: string;
  payload: Record<string, unknown>;
}

export class SSEClient {
  lastSeq = 0;
  private source: EventSource | null = null;
  constructor(private baseUrl: string) {}

  parse(raw: string): StreamEvent | null {
    const match = raw.match(/data: (.+)\n\n/);
    if (!match) return null;
    const event = JSON.parse(match[1]) as StreamEvent;
    this.lastSeq = event.seq;
    return event;
  }

  connect(sessionId: string, onEvent: (e: StreamEvent) => void): void {
    this.source = new EventSource(`${this.baseUrl}/sessions/${sessionId}/events?after_seq=${this.lastSeq}`);
    this.source.onmessage = (msg) => {
      const event = this.parse(msg.data);
      if (event) onEvent(event);
    };
    this.source.onerror = () => {
      this.source?.close();
      setTimeout(() => this.connect(sessionId, onEvent), 1000);
    };
  }

  disconnect(): void {
    this.source?.close();
    this.source = null;
  }
}
```

      const msgs = [...s.messages];
      const last = msgs[msgs.length - 1];
      if (last?.role === "assistant") {

        msgs.push({ role: "assistant", content: text });

      return { messages: msgs };

  sessions: [],

    set({ activeId: id });
    return id;
  },
  remove: async (id) => {
    await api.deleteSession(id);
    await get().load();
  },
  select: (id) => set({ activeId: id }),
}));
